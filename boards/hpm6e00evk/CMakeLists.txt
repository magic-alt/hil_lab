cmake_minimum_required(VERSION 3.13)

find_package(hpm-sdk REQUIRED HINTS $ENV{HPM_SDK_BASE})

set(HIL_PWM_PROFILE "20k_50_700ns" CACHE STRING "BBB B1 PWM stimulus profile")

set(PWM_FREQUENCY_HZ 20000)
set(PWM_DUTY_PERMILLE 500)
set(PWM_DEADTIME_NS 700)
set(PWM_PROFILE_ID 2)

if(HIL_PWM_PROFILE STREQUAL "20k_50_600ns")
    set(PWM_DEADTIME_NS 600)
    set(PWM_PROFILE_ID 1)
elseif(HIL_PWM_PROFILE STREQUAL "20k_50_700ns")
    set(PWM_DEADTIME_NS 700)
    set(PWM_PROFILE_ID 2)
elseif(HIL_PWM_PROFILE STREQUAL "20k_50_800ns")
    set(PWM_DEADTIME_NS 800)
    set(PWM_PROFILE_ID 3)
elseif(HIL_PWM_PROFILE STREQUAL "20k_5_700ns")
    set(PWM_DUTY_PERMILLE 50)
    set(PWM_PROFILE_ID 4)
elseif(HIL_PWM_PROFILE STREQUAL "20k_25_700ns")
    set(PWM_DUTY_PERMILLE 250)
    set(PWM_PROFILE_ID 5)
elseif(HIL_PWM_PROFILE STREQUAL "20k_75_700ns")
    set(PWM_DUTY_PERMILLE 750)
    set(PWM_PROFILE_ID 6)
elseif(HIL_PWM_PROFILE STREQUAL "20k_95_700ns")
    set(PWM_DUTY_PERMILLE 950)
    set(PWM_PROFILE_ID 7)
else()
    message(FATAL_ERROR "Unknown HIL_PWM_PROFILE=${HIL_PWM_PROFILE}")
endif()

project(hil_hpm6e_pwm_stimulus)

sdk_inc(inc)
sdk_app_src(src/main.c)
sdk_compile_definitions(-DPWM_FREQUENCY_HZ=${PWM_FREQUENCY_HZ}UL)
sdk_compile_definitions(-DPWM_DUTY_PERMILLE=${PWM_DUTY_PERMILLE}UL)
sdk_compile_definitions(-DPWM_DEADTIME_NS=${PWM_DEADTIME_NS}UL)
sdk_compile_definitions(-DPWM_PROFILE_ID=${PWM_PROFILE_ID}UL)

generate_ide_projects()
